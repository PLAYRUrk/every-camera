# every-camera — общее ядро. Нужно на любом узле.
#
# Целевая платформа: CentOS 7.8 (glibc 2.17) + Python 3.11.7, прямой доступ к PyPI.
#
# Верхние границы здесь не косметика. За ними начинаются колёса, собранные под
# glibc >= 2.28 (на CentOS 7 их не поставить), либо несовместимый ABI. Поднимать
# границу можно, но только после проверки на самой машине:
#
#     python3.11 -m pip download -r requirements-<камера>.txt \
#                -d /tmp/wheels --only-binary=:all:
#
# Всё, что не скачалось колесом, на CentOS 7 придётся собирать из исходников —
# системный gcc 4.8 этого не потянет для numpy и подобных.
#
# Ставится не этот файл, а файл своей роли — он подключает ядро сам:
#
#     pip install -r requirements-cannon.txt     # узел с камерой Canon
#     pip install -r requirements-sptt.txt       # узел с камерой SPTT
#     pip install -r requirements-infra.txt      # узел с камерой Infra
#     pip install -r requirements-asi.txt        # узел с камерой ASI
#     pip install -r requirements-japan.txt      # узел с камерой Japan
#     pip install -r requirements-sentry.txt     # узел с камерой Sentry
#     pip install -r requirements-gui.txt        # плюс GUI (gui_app/setup_app)
#     pip install -r requirements-observer.txt   # машина наблюдателя
#
# Узлу с одной камерой зависимости остальных не нужны.

numpy>=1.23.5,<2.0     # <2.0: astropy 5.x/6.x и opencv собраны под ABI numpy 1.x
Pillow>=9.5,<11.0      # JPEG/PNG/TIFF; в 11.x меняется базовый образ manylinux
